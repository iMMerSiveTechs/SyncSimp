Configuration Example

version: 1

app:
  name: My Awesome App
  bundleId: com.company.myapp
  platform: ios

plans:
  - id: premium_monthly
    displayName: Premium Monthly
    store: ios
    type: auto_renewable
    appleProductId: com.myapp.premium.monthly
    entitlement: premium
    duration: P1M
    price:
      currency: USD
      amount: 9.99
    introOffer:
      type: free_trial
      duration: P7D
    rc:
      offering: default
      packageId: monthly
